# Compute API — installed on top of requirements.txt.
#
# Deliberately small. This image already carries pandas, scipy, scikit-learn and
# XGBoost, so every extra dependency costs build time and container size on a
# platform that charges for both.

fastapi>=0.110
uvicorn[standard]>=0.27
# Postgres driver + pool. Sync on purpose: the work here is CPU-bound, so async
# would block the event loop anyway (see api/main.py).
psycopg[binary,pool]>=3.1
# Encrypts Strava tokens at rest.
cryptography>=42.0
# Hashes account passwords (argon2id). Ships wheels; no system library to add.
argon2-cffi>=23.1
# Supabase Storage is reached over its REST API rather than the SDK.
httpx>=0.27
# Multipart parsing — FastAPI needs it for the image upload endpoint (UploadFile).
python-multipart>=0.0.9
# Rasterizes an uploaded blog PDF into one PNG per page. A pure wheel (bundles its
# own PDFium build) — no poppler/ghostscript system package to add to the Docker
# image, unlike pdf2image.
pypdfium2>=4.30
# Encodes the rasterized pages as PNG.
Pillow>=10.0
